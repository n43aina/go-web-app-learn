package main import ( "html/template" "log" "net/http" ) const page = ` My Static Website

My Website

Welcome to My Static Website

Written by n43aina

This is a simple static website served using Go (Golang). The page contains HTML and CSS while Go handles the HTTP server.

Thanks for visiting!

© 2026 n43aina
` func handler(w http.ResponseWriter, r *http.Request) { tmpl := template.Must(template.New("page").Parse(page)) tmpl.Execute(w, nil) } func main() { http.HandleFunc("/", handler) log.Println("Server running at http://localhost:8080") log.Fatal(http.ListenAndServe(":8080", nil)) }
